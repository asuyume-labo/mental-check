function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    saveData(data);
    savePdfToDrive(data);         // ← Drive保存
    if (data.email) { sendResultEmail(data); }
    return ContentService
      .createTextOutput(JSON.stringify({status:"ok"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({status:"error", message:err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// ── Driveにレポートを保存 ────────────────────────────────────
function savePdfToDrive(data) {

  let folders = DriveApp.getFoldersByName(DRIVE_FOLDER_NAME);
  let folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(DRIVE_FOLDER_NAME);

  const name = data.name || "匿名";
  const date = data.date || new Date().toLocaleDateString("ja-JP");
  const rc =
    data.risk_label === "低リスク" ? "#10B981" :
    data.risk_label === "注意"     ? "#F59E0B" :
    data.risk_label === "要対応"   ? "#F97316" : "#EF4444";

    ["C:職場のサポート",     data.score_C],
    ["D:身体的・心理的反応", data.score_D],
    ["E:活気・仕事の満足感", data.score_E],
  ].map(([title, score]) =>
    `<tr><td style="padding:6px 10px;border-bottom:1px solid #eee;font-size:12px">${title}</td>
     <td style="padding:6px 10px;border-bottom:1px solid #eee;font-weight:bold;text-align:right;font-size:13px">${score}点</td></tr>`
  ).join("");

  // レポートHTML（Google DocのPDF変換用）
  const html = `<!DOCTYPE html>
<html lang="ja"><head><meta charset="UTF-8">
<style>
  body{font-family:'Hiragino Kaku Gothic ProN',Meiryo,sans-serif;font-size:11px;color:#1E293B;margin:20px;max-width:600px}
  .header{background:#0A0A0A;color:#D4AF37;padding:12px 16px;border-radius:8px;margin-bottom:14px}
  .header h1{font-size:16px;margin:0 0 2px}

<body>
<div class="header">

  <div class="info-item"><span class="info-label">性別：</span>${data.gender||"—"}</div>

  const fileName = `${date}_${name}_メンタルスクリーニング結果.pdf`;

  // HTMLをGoogleドキュメントとして一時保存
  const tempFile = DriveApp.createFile(blob);

  // PDF形式でエクスポート
  const pdfUrl = `https://docs.google.com/feeds/download/documents/export/Export?id=${docId}&exportFormat=pdf`;
  const response = UrlFetchApp.fetch(pdfUrl, {

    muteHttpExceptions: true
  });

  const pdfBlob = response.getBlob().setName(fileName);
  folder.createFile(pdfBlob);

  // 一時ファイル削除
  tempFile.setTrashed(true);
}

// ── スプレッドシート保存 ──────────────────────────────────────
function saveData(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName("回答データ");
  if (!sheet) sheet = ss.insertSheet("回答データ");

  if (sheet.getLastRow() === 0) {
    const row1 = ["受診日時","氏名","部署","年齢","性別","メール","総合スコア","リスク判定",
      "A:仕事負担","B:コントロール","C:サポート","D:身体心理","E:活気満足"];
    const row2 = ["","","","","","","","","","","","",""];
    QUESTIONS.forEach((q, i) => {
      row1.push("Q"+(i+1)+" 番号");
      row1.push("Q"+(i+1)+" 回答内容");
      row2.push("["+q.section+"] "+q.text);
      row2.push("");
    });
    sheet.appendRow(row1);
    sheet.appendRow(row2);
    const totalCols = row1.length;
    sheet.getRange(1,1,1,totalCols).setBackground("#1E293B").setFontColor("white").setFontWeight("bold").setHorizontalAlignment("center");
    sheet.getRange(2,1,1,totalCols).setBackground("#F1F5F9").setFontColor("#475569").setFontStyle("italic").setWrapStrategy(SpreadsheetApp.WrapStrategy.WRAP);
    sheet.setColumnWidth(1,150); sheet.setColumnWidth(2,100); sheet.setColumnWidth(3,100);
    sheet.setColumnWidth(4,60);  sheet.setColumnWidth(5,70);  sheet.setColumnWidth(6,180);
    sheet.setColumnWidth(7,80);  sheet.setColumnWidth(8,80);
    for(let i=9;i<=13;i++) sheet.setColumnWidth(i,70);
    for(let i=14;i<=totalCols;i++) sheet.setColumnWidth(i, i%2===0 ? 100 : 50);
    sheet.setRowHeight(2,80);
    sheet.setFrozenRows(2);
    sheet.setFrozenColumns(2);
  }

  const answers = data.answers || {};
  const row = [
    new Date().toLocaleString("ja-JP"),
    data.name||"", data.dept||"", data.age||"", data.gender||"", data.email||"",
    data.total_score, data.risk_label,
    data.score_A, data.score_B, data.score_C, data.score_D, data.score_E,
  ];
  QUESTIONS.forEach(q => {
    const ans = answers[q.key];
    row.push(ans && ans.answer ? ans.answer : "");
    row.push(ans && ans.answer ? ANSWER_LABELS[ans.answer]||"" : "");
  });
  sheet.appendRow(row);

  const lastRow = sheet.getLastRow();
  const bg = data.total_score>=75?"#FEE2E2":data.total_score>=60?"#FEF3C7":data.total_score>=40?"#FFFBEB":"#ECFDF5";
  sheet.getRange(lastRow,1,1,8).setBackground(bg);
}
